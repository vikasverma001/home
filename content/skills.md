<script>
(function () {
  var filters = document.getElementById('certFilters');
  if (!filters) return;
  var cards = Array.prototype.slice.call(document.querySelectorAll('#certBoard .cert-card'));
  var empty = document.getElementById('certEmpty');
  filters.addEventListener('click', function (e) {
    var btn = e.target.closest('.cert-pill');
    if (!btn) return;
    filters.querySelectorAll('.cert-pill').forEach(function (p) { p.classList.remove('active'); });
    btn.classList.add('active');
    var f = btn.getAttribute('data-filter');
    var shown = 0;
    cards.forEach(function (c) {
      var match = f === 'all' || c.getAttribute('data-cat') === f;
      c.style.display = match ? '' : 'none';
      if (match) shown++;
    });
    if (empty) empty.hidden = shown !== 0;
  });
})();
</script>
