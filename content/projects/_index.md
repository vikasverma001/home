(function () {
  var filters = document.getElementById("projFilters");
  if (!filters) return;
  var cards = Array.prototype.slice.call(document.querySelectorAll("#projGrid .proj-card"));
  var empty = document.getElementById("projEmpty");
  filters.addEventListener("click", function (e) {
    var btn = e.target.closest(".filter-pill");
    if (!btn) return;
    filters.querySelectorAll(".filter-pill").forEach(function (p) { p.classList.remove("active"); });
    btn.classList.add("active");
    var f = btn.getAttribute("data-filter");
    var shown = 0;
    cards.forEach(function (c) {
      var match = f === "all" || (" " + c.getAttribute("data-cat") + " ").indexOf(" " + f + " ") !== -1;
      c.style.display = match ? "" : "none";
      if (match) shown++;
    });
    if (empty) empty.hidden = shown !== 0;
  });
})();
</script>
